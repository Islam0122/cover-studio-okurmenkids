const universityBand: TemplateDefinition = {
  id: 'university-band',
  orientation: 'portrait',
  palette: { primary: '#0F1B3D', accent: '#4F7CFF', … },
  fonts: { heading: 'Montserrat', body: 'Inter' },
  layout: ({ width: W, height: H }) => {
    const band = W * 0.32
    return [
      shape('Боковая панель', 'rect',
        { x: 0, y: 0, width: band, height: H },
        { fill: '$primary' }),
      text('Учреждение', '{{institution}}', …),
    ]
  },
}

Deployment

A static single-page app

Vercel

npm run build → dist/. Every route is rewritten to index.html.

// vercel.json
{ "rewrites": [{
  "source": "/(.*)",
  "destination": "/index.html"
}] }

Netlify

The same SPA fallback, shipped from public/ into the build output.

# public/_redirects
/*  /index.html  200

Static hosts

For hosts without rewrites (e.g. GitHub Pages): hash routing and relative paths.

# npm run build:static
VITE_HASH_ROUTER=1 vite build \
  --base=./
# → /#/editor/:projectId

No production URL is published in the repository, so none is shown here.
